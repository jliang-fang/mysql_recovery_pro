## 六、测试数据生成

### 6.1 测试数据生成脚本

```python
# test/test_data_generator.py
import os
import struct

def generate_test_page(page_type, row_format, num_records):
    """生成测试页"""
    page = bytearray(16384)
    
    # FIL Header
    struct.pack_into('<I', page, 0, 0)      # FIL_PAGE_PREV
    struct.pack_into('<I', page, 4, 0)      # FIL_PAGE_NEXT
    struct.pack_into('<Q', page, 8, 0)      # FIL_PAGE_LSN
    struct.pack_into('<H', page, 16, page_type)  # FIL_PAGE_TYPE
    struct.pack_into('<I', page, 18, 0)     # FIL_PAGE_SPACE_ID
    struct.pack_into('<I', page, 20, 0)     # FIL_PAGE_OFFSET
    
    # Page Header
    struct.pack_into('<H', page, 38, 10)    # PAGE_N_DIR_SLOTS
    struct.pack_into('<H', page, 40, 100)   # PAGE_HEAP_TOP
    struct.pack_into('<H', page, 42, num_records + 2)  # PAGE_N_HEAP
    
    return page

def generate_test_file(num_pages):
    """生成测试文件"""
    with open('test.ibd', 'wb') as f:
        for i in range(num_pages):
            page = generate_test_page(0x45bf, 1, 100)
            f.write(page)
```
